cmake_minimum_required(VERSION 3.20)
project(collatz VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(COLLATZ_VISUAL_TESTS "Include headless Python visualization tests in CTest" OFF)
option(COLLATZ_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(COLLATZ_TSAN "Enable ThreadSanitizer (separate build)" OFF)
option(COLLATZ_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
option(COLLATZ_LTO "Enable supported link-time optimization in Release" ON)
if(COLLATZ_SANITIZERS AND COLLATZ_TSAN)
    message(FATAL_ERROR "ASan/UBSan and TSan require separate builds")
endif()

if(COLLATZ_LTO AND NOT COLLATZ_SANITIZERS AND NOT COLLATZ_TSAN)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT collatz_ipo_supported OUTPUT collatz_ipo_error LANGUAGES CXX)
    if(collatz_ipo_supported)
        set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
    else()
        message(STATUS "Release LTO unavailable; continuing without it: ${collatz_ipo_error}")
    endif()
endif()

find_package(Threads REQUIRED)
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_LIBRARIES Threads::Threads)
check_cxx_source_compiles("
    #include <stop_token>
    #include <thread>
    int main() {
        std::jthread worker([](std::stop_token stop) { (void)stop.stop_requested(); });
        worker.request_stop();
    }
" COLLATZ_HAS_JTHREAD)
unset(CMAKE_REQUIRED_LIBRARIES)
if(NOT COLLATZ_HAS_JTHREAD)
    message(FATAL_ERROR
        "C++20 std::jthread and std::stop_token are required. "
        "On macOS, use Xcode/Command Line Tools 26 or newer.")
endif()
include(cmake/VerifyJson.cmake)
add_library(collatz_options INTERFACE)
if(MSVC)
    target_compile_options(collatz_options INTERFACE /W4 /permissive- /utf-8)
    target_compile_definitions(collatz_options INTERFACE NOMINMAX)
    if(COLLATZ_WARNINGS_AS_ERRORS)
        target_compile_options(collatz_options INTERFACE /WX)
    endif()
    if(COLLATZ_SANITIZERS OR COLLATZ_TSAN)
        message(FATAL_ERROR "Sanitizer options require Clang or GCC")
    endif()
else()
    target_compile_options(collatz_options INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wold-style-cast)
    if(COLLATZ_WARNINGS_AS_ERRORS)
        target_compile_options(collatz_options INTERFACE -Werror)
    endif()
    if(COLLATZ_SANITIZERS)
        target_compile_options(collatz_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(collatz_options INTERFACE -fsanitize=address,undefined)
    elseif(COLLATZ_TSAN)
        target_compile_options(collatz_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
        target_link_options(collatz_options INTERFACE -fsanitize=thread)
    endif()
endif()

add_library(collatz_core src/core/collatz_core.cpp src/core/summary.cpp src/core/range_executor.cpp)
target_include_directories(collatz_core PUBLIC include PRIVATE src)
target_link_libraries(collatz_core PUBLIC Threads::Threads collatz_options)
add_library(collatz_config src/app/config.cpp)
target_include_directories(collatz_config PRIVATE src)
target_include_directories(collatz_config SYSTEM PRIVATE third_party)
target_link_libraries(collatz_config PUBLIC collatz_core)
configure_file(src/app/build_info.hpp.in generated/collatz_build_info.hpp @ONLY)
add_library(collatz_output src/app/output_bundle.cpp src/app/run_info.cpp src/app/exporters.cpp)
target_include_directories(collatz_output PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(collatz_output SYSTEM PRIVATE third_party)
target_compile_definitions(collatz_output PRIVATE COLLATZ_BUILD_CONFIGURATION="$<CONFIG>")
target_link_libraries(collatz_output PUBLIC collatz_config)
add_executable(collatz_run src/app/main.cpp)
target_link_libraries(collatz_run PRIVATE collatz_output)

add_executable(collatz_profile EXCLUDE_FROM_ALL benchmarks/profile.cpp)
target_include_directories(collatz_profile PRIVATE src)
target_link_libraries(collatz_profile PRIVATE collatz_core)

include(CTest)
if(BUILD_TESTING)
    foreach(name collatz_core memo_consistency summary digit_counter executor csv_writer deferred_summary doubling_families)
        add_executable(${name}_test tests/cpp/${name}_test.cpp)
        target_link_libraries(${name}_test PRIVATE collatz_core)
        target_include_directories(${name}_test PRIVATE src)
        add_test(NAME ${name} COMMAND ${name}_test)
    endforeach()
    add_executable(output_bundle_test tests/cpp/output_bundle_test.cpp)
    target_link_libraries(output_bundle_test PRIVATE collatz_output)
    target_include_directories(output_bundle_test PRIVATE src)
    add_test(NAME output_bundle COMMAND output_bundle_test)
    # Compile public headers with only their advertised include directory.
    set(header_checks)
    foreach(header collatz_core summary range_executor)
        set(check "${CMAKE_CURRENT_BINARY_DIR}/header_checks/${header}.cpp")
        file(GENERATE OUTPUT "${check}" CONTENT "#include \"collatz/${header}.hpp\"\n")
        list(APPEND header_checks "${check}")
    endforeach()
    add_library(collatz_header_check OBJECT ${header_checks})
    target_link_libraries(collatz_header_check PRIVATE collatz_core)
    if(COLLATZ_VISUAL_TESTS)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
    else()
        find_package(Python3 COMPONENTS Interpreter)
    endif()
    if(Python3_Interpreter_FOUND)
        add_test(NAME integration COMMAND "${Python3_EXECUTABLE}"
            "${PROJECT_SOURCE_DIR}/tests/python/integration_test.py" "$<TARGET_FILE:collatz_run>")
        add_test(NAME result_paths COMMAND "${Python3_EXECUTABLE}"
            "${PROJECT_SOURCE_DIR}/tests/python/result_paths_test.py")
    endif()
    if(COLLATZ_VISUAL_TESTS)
        add_test(NAME visuals COMMAND "${Python3_EXECUTABLE}" -W error::UserWarning
            "${PROJECT_SOURCE_DIR}/tests/python/visuals_test.py" "$<TARGET_FILE:collatz_run>")
        set_tests_properties(visuals PROPERTIES ENVIRONMENT "MPLBACKEND=Agg;PYTHONUTF8=1")
    endif()
elseif(COLLATZ_VISUAL_TESTS)
    message(FATAL_ERROR "COLLATZ_VISUAL_TESTS requires BUILD_TESTING=ON")
endif()
